// Mã C++ chuẩn nạp trực tiếp cho mạch Arduino Uno / ESP32 từ DrgX Dev Studio
#include <Arduino.h>
const int TRIG_PIN = 9, ECHO_PIN = 10, ENA = 5, ENB = 6, LED_OK = 11, LED_WARN = 13;

long readDistanceCm() {
  digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  return pulseIn(ECHO_PIN, HIGH, 25000) * 0.034 / 2;
}

void setMotors(int pwmL, int pwmR) {
  analogWrite(ENA, constrain(pwmL, 0, 255));
  analogWrite(ENB, constrain(pwmR, 0, 255));
}

void setup() {
  Serial.begin(115200);
  pinMode(TRIG_PIN, OUTPUT); pinMode(ECHO_PIN, INPUT);
  pinMode(ENA, OUTPUT); pinMode(ENB, OUTPUT);
  pinMode(LED_OK, OUTPUT); pinMode(LED_WARN, OUTPUT);
}

void loop() {
  long d = readDistanceCm();
  if (d < 15) { digitalWrite(LED_WARN, HIGH); setMotors(180, -180); } // Rẽ tránh vật cản theo đỉnh tiếp theo của BFS
  else { digitalWrite(LED_OK, HIGH); setMotors(220, 220); }
  delay(80);
}
